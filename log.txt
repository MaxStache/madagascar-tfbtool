            set [~found variable] = TAG0 - 1; 
            flow continue
        }
        set dying?::user::value = TAG0 - 1; 
        flow continue
    }
    //  
    check value ( dying?::user::value != TAG0 - 0 ) {
        set dying?::user::value = TAG0 - 0; 
        // Now, put the items back on. 
        if ( find variable ( global[ alex_actor_to_be_clothed::user::actor ] in global[ Zoovenirs_Master::actor ] ) ) {
            set reference *[~found variable] = my; 
            flow continue
        } else {
            print(my, "Could NOT set reference in ZOOVENIR master."); 
            flow continue
        }
        flow continue
    }
    //  
    check value ( global[ Script Time::value ] > (timer_darkness::user::value + TAG0 - 0) ) {
        setBehavior(Leader::behavior); 
        set dying?::user::value = TAG0 - 0; 
        check value ( global[ health_alex::user::value ] == TAG0 - 0 ) {
            set global[ health_alex::user::value ] = (global[ ZoosterHealthContainers::user::value ] + TAG0 - 0); 
            flow continue
        }
        flow continue
    }
    flow continue
}

        set my.tint color = Color(128, 128, 128, 255); 
        set state_dying::user::value = TAG0 - 0; 
        flow continue
    } else if ( global[ Script Time::value ] >= (timer_hurt_youch::user::value + TAG0 - 0) ) {
        // TS- now that the timer_hurt_youch has expired, release control back to the player 
        // TS- If we don't release control, the player will be moving while playing the hurt anim, which may look strange 
        playAnimation(ambient); 
        setBehavior(Leader::behavior); 
        flow continue
    } else {
        // TS- until the timer_hurt_youch has expired, play my hurt anim and prevent the player from controlling me 
        runAsPlayer(global[ no controller::controller ]); 
        playAnimation(extra 29); 
        flow continue
    }
    flow continue
}
